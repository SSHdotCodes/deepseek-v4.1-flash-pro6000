cmake_minimum_required(VERSION 3.24)
project(ds41_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(DS41_CUDA "Build native single-GPU forward" ON)
option(DS41_SPARK_NATIVE "Build owner native drafter candidate (runtime opt-in)" OFF)
option(DS41_SPARK_POLICY_V2 "Link host-only policy-v2 sidecar (runtime opt-in)" OFF)
set(DS41_BLAS_RUNTIME_DIR "" CACHE PATH "Explicit exact-fixture BLAS-only runtime pin")
if(DS41_BLAS_RUNTIME_DIR)
 set(DS41_CUBLAS_SHA256 "25cde432108543626df85d887c730bf2cab96803ca16f9654e4abc8d2b45833e")
 set(DS41_CUBLASLT_SHA256 "1489aeb75d55ee55782798343ac66f57771d561cc1fe1e0491568deefa0e6515")
 set(DS41_CUDART_SHA256 "7fb00519a24512bf677b59e87921e5c05aac8246ceb6a57435c0d9dfeef364f6")
 foreach(name libcublas.so.13 libcublasLt.so.13)
  file(SHA256 "${DS41_BLAS_RUNTIME_DIR}/lib/${name}" actual)
  if(name STREQUAL "libcublas.so.13")
   set(expected "${DS41_CUBLAS_SHA256}")
  else()
   set(expected "${DS41_CUBLASLT_SHA256}")
  endif()
  if(NOT actual STREQUAL expected)
   message(FATAL_ERROR "BLAS runtime differs from exact-fixture pin: ${name}")
  endif()
 endforeach()
 # Frozen candidates use their small symlink overlay. Builds use the same
 # authenticated installed pair, never a whole Torch CUDA library directory.
 list(PREPEND CMAKE_BUILD_RPATH "$ORIGIN/blas" "${DS41_BLAS_RUNTIME_DIR}/lib")
endif()
set(DS41_COMPONENT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party-hotpath" CACHE PATH "Immutable matched component snapshot")
set(DS41_MOE_SOURCE_API "2" CACHE STRING "Explicit source ABI of the matched MoE bundle (2 or 4)")
if(NOT DS41_MOE_SOURCE_API MATCHES "^(2|4)$")
 message(FATAL_ERROR "DS41_MOE_SOURCE_API must match a frozen API2 or candidate API4 bundle")
endif()
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(ds41_spark SHARED third_party/spark/src/sampling.cpp third_party/spark/src/policy.cpp)
target_include_directories(ds41_spark PUBLIC third_party/spark/include)
target_compile_options(ds41_spark PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra>)
add_library(engine_core src/store.cpp src/sampling.cpp src/speculation.cpp src/server.cpp src/target_shared.cpp)
target_include_directories(engine_core PUBLIC include vendor)
target_compile_options(engine_core PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(engine_core PUBLIC Threads::Threads ds41_spark OpenSSL::Crypto)
add_executable(ds41d src/main.cpp)
target_link_libraries(ds41d PRIVATE engine_core)
if(DS41_CUDA)
 enable_language(CUDA)
 find_package(CUDAToolkit REQUIRED)
 set(DS41_AUXILIARY_FILE "${CMAKE_CURRENT_SOURCE_DIR}/generated/auxiliary.json" CACHE FILEPATH "Authenticated auxiliary receipt")
 if(NOT EXISTS "${DS41_AUXILIARY_FILE}")
  message(FATAL_ERROR "Missing ${DS41_AUXILIARY_FILE}: run scripts/prepare_assets.py, then scripts/build.sh --assets DIR")
 endif()
 file(SHA256 "${DS41_AUXILIARY_FILE}" DS41_AUX_SHA256)
 foreach(component moe dense attn)
  if(NOT EXISTS "${DS41_COMPONENT_DIR}/lib/libds41_${component}.so")
   message(FATAL_ERROR "Missing component ${component} in ${DS41_COMPONENT_DIR}: build it with scripts/build.sh")
  endif()
  add_library(ds41_${component} SHARED IMPORTED)
  set_target_properties(ds41_${component} PROPERTIES IMPORTED_LOCATION "${DS41_COMPONENT_DIR}/lib/libds41_${component}.so")
 endforeach()
 if(DS41_SPARK_NATIVE)
  target_sources(ds41_spark PRIVATE third_party/spark/src/forward.cu third_party/spark/src/sampling.cu)
  target_include_directories(ds41_spark PRIVATE "${DS41_COMPONENT_DIR}/include" third_party/include)
  target_link_libraries(ds41_spark PRIVATE ds41_dense ds41_attn ds41_moe CUDA::cudart CUDA::cublas)
  target_compile_options(ds41_spark PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;--fmad=false;-lineinfo>)
  set_target_properties(ds41_spark PROPERTIES CUDA_ARCHITECTURES "120a" CUDA_STANDARD 17)
 endif()
 add_library(engine_ops SHARED src/ops.cu src/hotpath/kernels.cu src/prefill/kernels.cu)
 target_include_directories(engine_ops PUBLIC include)
 target_link_libraries(engine_ops PUBLIC CUDA::cudart)
 # Batched prefill row kernels (src/prefill/kernels.cu) call the dense cuBLAS wrapper and cuBLAS directly.
 target_include_directories(engine_ops PRIVATE "${DS41_COMPONENT_DIR}/include")
 target_link_libraries(engine_ops PRIVATE ds41_dense CUDA::cublas)
 target_compile_options(engine_ops PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;--fmad=false;-lineinfo>)
 set_target_properties(engine_ops PROPERTIES CUDA_ARCHITECTURES "120a" CUDA_STANDARD 17)
 add_library(engine_cuda src/forward.cu src/vision.cu src/engram_async.cpp src/spark_bridge.cpp src/prefill/stager.cpp src/blas_runtime.cpp)
 if(DS41_BLAS_RUNTIME_DIR)
  target_compile_definitions(engine_cuda PRIVATE DS41_BLAS_PIN_ACTIVE=1
      DS41_CUBLAS_SHA256="${DS41_CUBLAS_SHA256}"
      DS41_CUBLASLT_SHA256="${DS41_CUBLASLT_SHA256}"
      DS41_CUDART_SHA256="${DS41_CUDART_SHA256}")
 endif()
 if(DS41_SPARK_POLICY_V2)
  if(NOT DS41_SPARK_NATIVE)
   message(FATAL_ERROR "DS41_SPARK_POLICY_V2 requires DS41_SPARK_NATIVE")
  endif()
  # The policy sidecar is built from components/spark-policy into the component prefix. Its header must be the
  # frozen policy-v2 header.
  set(policy_root "${DS41_COMPONENT_DIR}")
  if(NOT EXISTS "${policy_root}/lib/libds41_spark_policy_v2.so")
   message(FATAL_ERROR "Missing libds41_spark_policy_v2.so in ${policy_root}: build it with scripts/build.sh")
  endif()
  file(SHA256 "${policy_root}/include/ds41/spark_policy.h" policy_header_sha)
  if(NOT policy_header_sha STREQUAL "800275d3066460aab0f3f1a3122c40830e9cf2660b21e876370b811d00d7dadb")
   message(FATAL_ERROR "spark_policy.h differs from the policy-v2 freeze")
  endif()
  add_library(ds41_spark_policy_v2 SHARED IMPORTED)
  set_target_properties(ds41_spark_policy_v2 PROPERTIES IMPORTED_LOCATION "${policy_root}/lib/libds41_spark_policy_v2.so")
  target_include_directories(engine_cuda PRIVATE "${policy_root}/include")
  target_link_libraries(engine_cuda PRIVATE ds41_spark_policy_v2)
  target_compile_definitions(engine_cuda PRIVATE DS41_SPARK_POLICY_V2=1)
  configure_file("${policy_root}/lib/libds41_spark_policy_v2.so"
                 "${CMAKE_CURRENT_BINARY_DIR}/libds41_spark_policy_v2.so" COPYONLY)
 endif()
 if(DS41_SPARK_NATIVE)
  target_compile_definitions(engine_cuda PRIVATE DS41_SPARK_NATIVE=1)
 endif()
 target_include_directories(engine_cuda PUBLIC include vendor "${DS41_COMPONENT_DIR}/include" third_party/include)
 target_link_libraries(engine_cuda PUBLIC engine_ops engine_core ds41_moe ds41_dense ds41_attn CUDA::cudart CUDA::cublas ${CMAKE_DL_LIBS})
 target_compile_options(engine_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;--fmad=false;-lineinfo>)
 target_compile_definitions(engine_cuda PRIVATE DS41_MOE_SOURCE_API=${DS41_MOE_SOURCE_API} DS41_AUX_SHA256="${DS41_AUX_SHA256}" DS41_DEFAULT_AUXILIARY="${DS41_AUXILIARY_FILE}")
 set_target_properties(engine_cuda PROPERTIES CUDA_ARCHITECTURES "120a" CUDA_STANDARD 17)
 target_compile_definitions(ds41d PRIVATE DS41_CUDA=1)
 target_link_libraries(ds41d PRIVATE engine_cuda)
 add_executable(test_native_prefill tests/native_prefill.cpp)

 add_executable(test_native_state tests/native_state.cpp)

 add_executable(test_native_services tests/native_services.cpp)
 target_link_libraries(test_native_services PRIVATE engine_cuda)
 if(DS41_SPARK_NATIVE)
  add_executable(test_native_spark_seed tests/native_spark_seed.cpp)
  target_link_libraries(test_native_spark_seed PRIVATE engine_cuda)
 endif()
endif()
