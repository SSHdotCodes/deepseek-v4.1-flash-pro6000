# Expert store (VRAM slots, exclusive RAM tier, O_DIRECT/GDS NVMe reads, MXFP4 kernels).
# Library target identical to the original build; its benchmarks and evidence tools are not part of this repo.
cmake_minimum_required(VERSION 3.24)
project(ds41_moe LANGUAGES CXX)
option(DS41_CUDA "Build CUDA store and MXFP4 kernels" ON)
set(CMAKE_CXX_STANDARD 17)
add_library(ds41_moe SHARED src/store.cpp src/direct_reader.cpp src/manifest.cpp)
target_include_directories(ds41_moe PUBLIC include)
target_compile_options(ds41_moe PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra>)
find_package(Threads REQUIRED)
target_link_libraries(ds41_moe PRIVATE Threads::Threads)
if(DS41_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  target_sources(ds41_moe PRIVATE src/mxfp4.cu src/offload.cpp src/decode_backend.cpp src/gds_reader.cpp)
  target_compile_definitions(ds41_moe PRIVATE DS41_CUDA)
  target_link_libraries(ds41_moe PUBLIC CUDA::cudart PRIVATE ${CMAKE_DL_LIBS})
  set_target_properties(ds41_moe PROPERTIES CUDA_ARCHITECTURES "120a" CUDA_STANDARD 17)
  target_compile_options(ds41_moe PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;--fmad=false;-lineinfo>)
endif()
install(TARGETS ds41_moe LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
