cmake_minimum_required(VERSION 3.24)
project(ds41_dense LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
add_library(ds41_dense SHARED src/dense.cu src/ops.cu src/compact.cu)
target_include_directories(ds41_dense PUBLIC include)
target_link_libraries(ds41_dense PUBLIC CUDA::cudart CUDA::cublas)
set_target_properties(ds41_dense PROPERTIES CUDA_ARCHITECTURES "120a" CUDA_STANDARD 17 CXX_STANDARD 17)
target_compile_options(ds41_dense PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;--fmad=false;-lineinfo>)
find_package(Threads REQUIRED)
add_library(ds41_engram SHARED engram/src/store.cpp engram/src/dequant.cu)
target_include_directories(ds41_engram PUBLIC include PRIVATE engram/vendor)
target_link_libraries(ds41_engram PRIVATE CUDA::cudart Threads::Threads)
set_target_properties(ds41_engram PROPERTIES CUDA_ARCHITECTURES "120a" CUDA_STANDARD 17 CXX_STANDARD 17)
target_compile_options(ds41_engram PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;--fmad=false> $<$<COMPILE_LANGUAGE:CXX>:-O3;-Wall;-Wextra>)

install(TARGETS ds41_dense ds41_engram LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
