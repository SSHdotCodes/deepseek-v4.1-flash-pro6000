# Fused BF16 LM head: host dispatcher (head_decode.cpp) plus two CUDA GEMV kernel libraries.
# ds41d dlopens libds41_fuse_head.so when started with --fused-head.
cmake_minimum_required(VERSION 3.24)
project(ds41_fuse_head LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
find_package(CUDAToolkit REQUIRED)
set(DS41_FUSE_HEAD_CUDA_FLAGS "-O3 -lineinfo" CACHE STRING "nvcc flags for the head kernels")
foreach(kernel half_tree half_tune)
  if(kernel STREQUAL "half_tree")
    set(target ds41_fuse_head_half)
  else()
    set(target ds41_fuse_head_half_tune)
  endif()
  add_library(${target} SHARED src/head_${kernel}.cu)
  set_target_properties(${target} PROPERTIES CUDA_ARCHITECTURES "120a" CUDA_STANDARD 17)
  separate_arguments(flags NATIVE_COMMAND "${DS41_FUSE_HEAD_CUDA_FLAGS}")
  target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:${flags}>)
endforeach()
add_library(ds41_fuse_head SHARED src/head_decode.cpp)
target_include_directories(ds41_fuse_head PUBLIC include PRIVATE ${CUDAToolkit_INCLUDE_DIRS})
target_compile_options(ds41_fuse_head PRIVATE -O3)
target_link_libraries(ds41_fuse_head PRIVATE ds41_fuse_head_half ds41_fuse_head_half_tune)
install(TARGETS ds41_fuse_head ds41_fuse_head_half ds41_fuse_head_half_tune LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
